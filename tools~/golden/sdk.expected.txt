===== Arcade/Data
namespace Monetization.Runtime.Analytics
{
    public class ArcadeLevelStarted : AnalyticsEventData
    {
        public int Level { get; }
        public string LevelType { get; }

        public ArcadeLevelStarted(int level, string levelType)
        {
            Level = level;
            LevelType = levelType;
        }
    }

    public class ArcadeLevelFinished : AnalyticsEventData
    {
        public bool Success { get; }
        public float Duration { get; }
        public string Reward { get; }

        public ArcadeLevelFinished(bool success, float duration, string reward)
        {
            Success = success;
            Duration = duration;
            Reward = reward;
        }
    }

    public class ArcadeAppBackgrounded : AnalyticsEventData
    {
    }

}

===== Arcade/Implementation
using UnityEngine;

namespace Monetization.Runtime.Analytics
{
    [CreateAssetMenu(fileName = "ArcadeAnalytics", menuName = "GDMonetization/MeticaAnalytics/ScriptSO/Arcade")]
    public class ArcadeAnalytics : GDMeticaAnalytics, IArcadeAnalytics
    {
        public void LevelStarted(ArcadeLevelStarted data)
        {
            var payload = CreateBaseEvent();
            payload["Level"] = data.Level;
            payload["LevelType"] = data.LevelType;
            MergeCustomFields(payload, data);
            LogCustomEvent("levelStarted", payload);
        }

        public void LevelFinished(ArcadeLevelFinished data)
        {
            var payload = CreateBaseEvent();
            payload["Success"] = data.Success;
            payload["Duration"] = data.Duration;
            payload["reward_Amount"] = data.Reward;
            MergeCustomFields(payload, data);
            LogCustomEvent("levelFinished", payload);
        }

        public void AppBackgrounded(ArcadeAppBackgrounded data)
        {
            var payload = CreateBaseEvent();
            MergeCustomFields(payload, data);
            LogCustomEvent("appBackgrounded", payload);
        }

    }
}

===== Arcade/Extra ArcadeAnalyticsSO.cs
using UnityEngine;

namespace Monetization.Runtime.Analytics
{
    [CreateAssetMenu(fileName = "ArcadeAnalyticsSO", menuName = "GDMonetization/MeticaAnalytics/GenreSO/Arcade")]
    public class ArcadeAnalyticsSO : AnalyticsGenreSO
    {
        [SerializeField] private ArcadeAnalytics _arcadeAnalytics;

        public override GDMeticaAnalytics GetAnalytics() => _arcadeAnalytics ??= new ArcadeAnalytics();
    }
}
