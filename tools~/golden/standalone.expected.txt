        }

    }

}

===== Arcade/Implementation
using UnityEngine;

namespace GameDistrict.MeticaAnalytics
{
    [CreateAssetMenu(fileName = "ArcadeAnalytics", menuName = "MeticaAnalytics/Arcade")]
    public class ArcadeAnalytics : GDMeticaAnalytics, IArcadeAnalytics
    {
        private static ArcadeAnalytics _instance;
        public static ArcadeAnalytics Instance =>
            _instance != null ? _instance : (_instance = Resources.Load<ArcadeAnalytics>("ArcadeAnalytics"));

        public void LevelStarted(ArcadeLevelStarted data)
        {
            var payload = CreateBaseEvent();
            payload["Level"] = data.Level;
            payload["LevelType"] = data.LevelType;
            MergeCustomFields(payload, data);
            LogCustomEvent("levelStarted", payload);
        }

        public void LevelFinished(ArcadeLevelFinished data)
        {
            var payload = CreateBaseEvent();
            payload["Success"] = data.Success;
            payload["Duration"] = data.Duration;
            payload["reward_Amount"] = data.Reward;
            MergeCustomFields(payload, data);
            LogCustomEvent("levelFinished", payload);
        }

        public void AppBackgrounded(ArcadeAppBackgrounded data)
        {
            var payload = CreateBaseEvent();
            MergeCustomFields(payload, data);
            LogCustomEvent("appBackgrounded", payload);
        }

    }
}
